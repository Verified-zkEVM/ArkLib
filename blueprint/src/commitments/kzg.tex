\section{KZG}\label{sec:kzg}

ArkLib formalizes the KZG polynomial commitment scheme~\cite{KZG10} in additive notation over a
finite field $\mathbb{F}$. Fix $\mathbb{F}$-modules $G_1,G_2,G_T$ with elements $g_1\in G_1$ and
$g_2\in G_2$, and an $\mathbb{F}$-bilinear pairing $e:G_1\times G_2\to G_T$. The SDH-based security
statements below assume $e(g_1,g_2)\neq 0$ and that $a\mapsto a\cdot g_1$ is surjective. Fix a maximal
degree $n$. The committed data is a coefficient vector $(a_0,\ldots,a_n)\in\mathbb{F}^{n+1}$,
that is, a polynomial $f$ of degree at most $n$; a query is a point $q\in\mathbb{F}$ and its
response is $f(q)$. Only $\mathbb{F}=\mathbb{Z}_r$ for the prime order $r$ of the groups is
realizable in practice.

In the algebraic group model, ArkLib also proves extractability, evaluation binding, and function
binding by computable reductions to $(n,1)$-PairingDLog with additive error $1/|\mathbb{F}|$,
assuming only $e(g_1,g_2)\neq 0$ and a complete polynomial root finder.
These bounds include $n=0$, and CompPoly's executable finite-field root finder satisfies the
completeness contract under its splitter-validity hypothesis.
The randomized bounds also instantiate CompPoly's Las Vegas splitter with sampled probe tables;
enumeration fallback gives unconditional completeness, with no running-time claim.

In the Type~3 generic group model, both existing binding games have advantage at most
$((n+1)(q+2n+3)(q+2n+4)+2)/|\mathbb{F}|$, assuming only $e(g_1,g_2)\neq 0$, with no
generator-surjectivity assumption. Root finding is scalar work, so the proof can use field
enumeration without changing the operation count.
The FKL translation followed by recovery from the first inconsistent claim uses at most
$q+n+1$ group operations, independently of the number of openings.

\begin{definition}[KZG commitment scheme]\label{def:kzg}
\lean{KZG.committerKey, KZG.verifierKey, KZG.keygen, KZG.commit, KZG.generateOpening,
KZG.verifyOpening}
\leanok
\begin{itemize}
  \item \textbf{Key generation.} Sample a uniform nonzero $\tau\in\mathbb{F}$. The committer key
    is $(\tau^i\cdot g_1)_{i\leq n}$ and the verifier key is $(g_2,\tau\cdot g_2)$.
  \item \textbf{Commitment.} $C=\sum_{i\leq n}a_i\cdot(\tau^i\cdot g_1)=f(\tau)\cdot g_1$.
  \item \textbf{Opening.} To open at $q$ with value $r$, the prover sends one element
    $\pi\in G_1$, the commitment to the quotient $(f(X)-f(q))/(X-q)$.
  \item \textbf{Verification.} Accept when $e(C-r\cdot g_1,g_2)=e(\pi,\tau\cdot g_2-q\cdot g_2)$.
\end{itemize}
\end{definition}

\begin{theorem}[KZG correctness]\label{thm:kzg_correctness}
\lean{KZG.verifyOpening_generateOpening}
\leanok
\uses{def:kzg}
For every trapdoor $\tau$, coefficient vector and point $q$, the verifier accepts the honest
commitment, the value $f(q)$ and the honest opening proof.
\end{theorem}

\begin{lemma}[Opening equation]\label{lem:kzg_opening_eq}
\lean{KZG.verifyOpening_eq}
\leanok
\uses{def:kzg}
Assume $e(g_1,g_2)\neq 0$ and that $g_1$ generates $G_1$. If the verifier accepts $(C,q,r,\pi)$
under the verifier key for $\tau$, then $C-r\cdot g_1=(\tau-q)\cdot\pi$.
\end{lemma}

\begin{theorem}[KZG evaluation binding, pointwise]\label{thm:kzg_binding}
\lean{KZG.evalBindingReduction, KZG.evalBindingReduction_wins, TSDH.experiment}
\leanok
\uses{def:kzg,lem:kzg_opening_eq}
A $t$-SDH adversary receives the keys and outputs $(c,h)\in\mathbb{F}\times G_1$; it wins when
$\tau+c\neq 0$ and $h=(\tau+c)^{-1}\cdot g_1$. Assume $e(g_1,g_2)\neq 0$ and that $g_1$ generates
$G_1$. If two openings $(r_1,\pi_1)$ and $(r_2,\pi_2)$ of one commitment at one point $q$ are
both accepted and $r_1\neq r_2$, then $(-q,(r_2-r_1)^{-1}\cdot(\pi_1-\pi_2))$ wins.
\end{theorem}

\begin{corollary}[KZG evaluation binding from $t$-SDH]\label{cor:kzg_binding_game}
\lean{KZG.evalBindingExperiment, KZG.evalBindingAdversary, KZG.evalBinding_le_tsdh,
KZG.evalBinding_le_of_tsdh}
\leanok
\uses{thm:kzg_binding}
In the evaluation-binding game the trapdoor $\tau$ is sampled uniformly from $\mathbb{F}^\times$,
and the adversary $\mathcal{A}$ receives the keys and wins when it outputs two accepted openings
of one commitment at one point with different values. Under the assumptions of
Theorem~\ref{thm:kzg_binding}, the probability that $\mathcal{A}$ wins is at most the
probability that the $t$-SDH adversary $\mathcal{B}$ wins, where $\mathcal{B}$ runs
$\mathcal{A}$ once on its keys and applies the reduction of Theorem~\ref{thm:kzg_binding}. If
every adversary in a class containing $\mathcal{B}$ wins the $t$-SDH game with probability at
most $\varepsilon$, then $\mathcal{A}$ wins with probability at most $\varepsilon$.
\end{corollary}

\begin{theorem}[KZG function binding, pointwise]\label{thm:kzg_function_binding}
\lean{KZG.FunctionBinding.functionBindingReduction,
KZG.FunctionBinding.functionBindingReduction_wins, ARSDH.experiment}
\leanok
\uses{def:kzg,lem:kzg_opening_eq,thm:kzg_binding}
An ARSDH adversary~\cite{CGKY25} receives the keys and outputs a set $S\subseteq\mathbb{F}$ and
$h_1,h_2\in G_1$. With $Z_S(X)=\prod_{s\in S}(X-s)$, it wins when $|S|=n+1$, $Z_S(\tau)\neq 0$,
$h_1\neq 0$ and $h_2=Z_S(\tau)^{-1}\cdot h_1$. Assume $e(g_1,g_2)\neq 0$ and that $g_1$
generates $G_1$, and fix $n+2$ distinct points of $\mathbb{F}$. If every opening in a list is
accepted for a commitment and no polynomial of degree at most $n$ takes all the opened values,
then an explicit reduction computes a winning ARSDH answer with $O(n+L)$ group operations for
$L$ openings.
\end{theorem}

\begin{corollary}[KZG function binding from ARSDH]\label{cor:kzg_function_binding_game}
\lean{KZG.functionBindingExperiment, KZG.functionBindingAdversary,
KZG.functionBinding_le_arsdh, KZG.functionBinding_le_of_arsdh}
\leanok
\uses{thm:kzg_function_binding}
In the function-binding game the trapdoor is sampled as above, and the adversary $\mathcal{A}$
wins when it outputs a commitment and a list of accepted openings that no polynomial of degree at
most $n$ fits. Under the assumptions of Theorem~\ref{thm:kzg_function_binding}, the probability
that $\mathcal{A}$ wins is at most the probability that the ARSDH adversary $\mathcal{B}$ wins,
where $\mathcal{B}$ runs $\mathcal{A}$ once on its keys and applies the reduction of
Theorem~\ref{thm:kzg_function_binding}. If every adversary in a class containing $\mathcal{B}$
wins the ARSDH game with probability at most $\varepsilon$, then $\mathcal{A}$ wins with
probability at most $\varepsilon$.
\end{corollary}
